Servicio de Notificaciones Toast

Métodos Básicos

ToastService.success('¡Operación exitosa!');
ToastService.error('Ha ocurrido un error');
ToastService.warning('Advertencia importante');
ToastService.info('Información útil');

Con Opciones Personalizadas

// Toast personalizado con duración específica
ToastService.show('Mensaje personalizado', 'success', {
  duration: 5000,
  icon: 'fa-heart'
});

// Toast sin auto-cerrar
ToastService.success('Mensaje permanente', {
  duration: 0,
  dismissible: true
});

Mensajes Múltiples

// Mostrar múltiples toasts
ToastService.success('Primer mensaje');
ToastService.info('Segundo mensaje');
ToastService.warning('Tercer mensaje');

// Limpiar todos los toasts
ToastService.clear();

Configuración Global

// Cambiar configuración global
ToastService.configure({
  position: 'top-right',
  duration: 5000,
  maxToasts: 3,
  offset: {
    vertical: 30,
    horizontal: 30,
    spacing: 15
  }
});

Uso en AJAX (Ejemplo de tu código)

$.ajax({
  url: '/api/guardar',
  type: 'POST',
  data: { ... },
  success: function(response) {
    if (response.status === 'success') {
      ToastService.success('Recepción guardada correctamente.');
    } else {
      ToastService.error('Error: ' + response.message);
    }
  },
  error: function() {
    ToastService.error('Error de comunicación con el servidor.');
  }
});

Inclusión en tu Proyecto

<!-- Incluir el servicio de toast en tu HTML -->
<script src="/js/toast-service.js"></script>

<!-- En tus archivos .phtml de Zend Framework 3 -->
<?php echo $this->headScript()->appendFile($this->basePath('js/toast-service.js')); ?>
